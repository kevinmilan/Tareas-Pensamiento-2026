suma_columnas = [0,0,0] # guarda la suma de cada columna


for i in range(3): # for por las filas, son 3
    fila = [] # creamos una lista para guardar cada fila
    for j in range(3): # este for es para las columnas, son 3 tambien columnas
        num = int(input("Ingrese número [{i}][{j}]: ")) # pedimos al usuario cada numero
        fila.append(num) # guardamos el numero en la fila
        suma_filas[i] += num # se suma ese numero en su fila, en suma_filas[0]
        suma_columnas[j] += num # se suma tambien ese numero a su columna, en suma_columnas
    matriz.append(fila)
print("Suma de filas: ", suma_filas)
